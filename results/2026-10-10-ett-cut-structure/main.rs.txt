//! Isolated ETT structural replay; intentionally records no elapsed times.
use knotrel_benchmarks::{Config, Operation, WorkloadKind, generate_with_query_percent};
use knotrel_core::{ForestGraph, ForestStats, diagnostics};
use serde_json::json;
fn stats(s: ForestStats) -> [u64; 4] {
    [s.tree_cuts, s.scanned_vertices, s.candidate_edges, s.replacements]
}
fn main() -> Result<(), Box<dyn std::error::Error>> {
    let args: Vec<_> = std::env::args().collect();
    let kind = WorkloadKind::parse(args.get(1).ok_or("workload required")?)?;
    let config = Config { nodes: args.get(2).ok_or("nodes required")?.parse()?, rounds: 10, seed: 42 };
    let query_percent = args.get(3).ok_or("query percent required")?.parse()?;
    let workload = generate_with_query_percent(kind, config, query_percent)?;
    #[derive(serde::Serialize)]
    struct Trace<'a> {
        schema_version: u32, workload: &'static str, config: Config,
        #[serde(skip_serializing_if = "Option::is_none")]
        query_percent: Option<u8>, trace: &'a knotrel_benchmarks::Workload,
    }
    let bytes = serde_json::to_vec(&Trace { schema_version: 1, workload: kind.name(), config,
        query_percent: kind.is_sustained().then_some(query_percent), trace: &workload })?;
    let fingerprint = bytes.iter().fold(0xcbf29ce484222325_u64, |h,b| (h ^ u64::from(*b)).wrapping_mul(0x100000001b3));
    let mut graph = ForestGraph::default();
    for v in 0..config.nodes { assert!(graph.add_node(v)); }
    for &(a,b) in &workload.initial_edges { assert!(graph.link(a,b)?); }
    graph.reset_stats(); diagnostics::take();
    let mut operations = Vec::new();
    for (index, op) in workload.operations.iter().enumerate() {
        diagnostics::take();
        let before = stats(graph.stats());
        let (name, answer) = match *op {
            Operation::Link {source,target} => { let answer=graph.link(source,target)?; assert!(answer); ("link",answer) }
            Operation::Cut {source,target} => { let answer=graph.cut(source,target)?; assert!(answer); ("cut",answer) }
            Operation::Connected {source,target,expected} => { let answer=graph.connected(source,target)?; assert_eq!(answer,expected); ("connected",answer) }
        };
        let counters = diagnostics::take();
        let after=stats(graph.stats());
        let delta: Vec<_> = after.iter().zip(before).map(|(a,b)|a-b).collect();
        operations.push(json!({"index":index,"operation":name,"answer":answer,"phases":counters,"forest_stats_delta":delta}));
    }
    println!("{}", serde_json::to_string_pretty(&json!({"workload":kind.name(),"nodes":config.nodes,"query_percent":query_percent,
        "trace_fingerprint_fnv1a64":format!("{fingerprint:016x}"),"operations":operations,"forest_stats":stats(graph.stats()),
        "phases":["other","detach","search","repair"],"columns":["root_calls","root_steps","rank_calls","rank_steps","split_entries","join_entries","pulls","rotations","candidate_tokens","mark_visits"]}))?);
    Ok(())
}
