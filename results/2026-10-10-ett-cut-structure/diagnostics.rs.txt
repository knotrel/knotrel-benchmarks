//! Isolated thread-local structural counters; not a production API or timer.
use std::cell::{Cell, RefCell};
thread_local! {
    static COUNTS: RefCell<[[u64; 10]; 4]> = const { RefCell::new([[0; 10]; 4]) };
    static PHASE: Cell<usize> = const { Cell::new(0) };
}
/// Restores the caller's phase, including unwinding and nested scopes.
pub struct Phase(usize);
impl Drop for Phase {
    fn drop(&mut self) {
        PHASE.with(|p| p.set(self.0));
    }
}
/// Enters other (0), detach (1), search (2), or replacement repair (3).
/// Panics for an invalid phase.
pub fn enter(phase: usize) -> Phase {
    assert!(phase < 4);
    Phase(PHASE.with(|p| p.replace(phase)))
}
/// Increments one structural counter in the active phase.
/// Columns: root calls/steps, rank calls/steps, split, join, pull,
/// rotations, candidate iterator tokens, candidate-mark ancestor visits.
/// Panics for an invalid counter. Each update has constant diagnostic overhead.
pub fn bump(counter: usize) {
    PHASE.with(|p| COUNTS.with(|c| c.borrow_mut()[p.get()][counter] += 1));
}
/// Reads and clears counters, preserving the active phase.
pub fn take() -> [[u64; 10]; 4] {
    COUNTS.with(|c| c.replace([[0; 10]; 4]))
}
#[cfg(test)]
mod tests {
    use super::*;
    #[test]
    fn nested_phases_restore_on_unwind_and_reset_counts() {
        take();
        bump(0);
        {
            let _a = enter(1);
            bump(1);
            let _ = std::panic::catch_unwind(|| {
                let _b = enter(2);
                bump(2);
                panic!("probe");
            });
            bump(1);
        }
        bump(0);
        let counts = take();
        assert_eq!(counts[0][0], 2);
        assert_eq!(counts[1][1], 2);
        assert_eq!(counts[2][2], 1);
        assert_eq!(take(), [[0; 10]; 4]);
    }
}
