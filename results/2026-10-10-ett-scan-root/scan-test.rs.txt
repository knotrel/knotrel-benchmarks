#[cfg(test)]
mod scan_root_tests {
    use super::Forest;
    #[test]
    fn membership_includes_unmarked_vertices_and_refreshes_after_mutations() {
        let mut f = Forest::default();
        for _ in 0..8 { f.add_vertex(); }
        let ab = f.link(0, 1); f.link(1, 2);
        f.link(3, 4); f.link(4, 5); f.link(5, 6);
        f.set_candidate(0, true);
        {
            let mut scan = f.smaller_candidates(0, 3);
            assert!(scan.contains_vertex(2)); // Unmarked member still belongs.
            assert!(!scan.contains_vertex(3)); assert!(!scan.contains_vertex(7));
            assert_eq!(scan.next(), Some(0)); assert_eq!(scan.next(), None);
            assert!(scan.contains_vertex(1)); // Exhaustion does not invalidate root.
        }
        f.cut(ab);
        {
            let scan = f.smaller_candidates(0, 1);
            assert!(scan.contains_vertex(0)); assert!(!scan.contains_vertex(1));
        }
        f.link(0, 3); // Rotate/merge, then choose the other component.
        {
            let scan = f.smaller_candidates(0, 1);
            assert!(scan.contains_vertex(2)); assert!(!scan.contains_vertex(0));
        }
        f.link(1, 7); // The smaller component grows but remains smaller.
        let scan = f.smaller_candidates(1, 0);
        assert!(scan.contains_vertex(7)); assert!(!scan.contains_vertex(3));
        f.validate();
    }
}
